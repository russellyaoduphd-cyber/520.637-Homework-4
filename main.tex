\subsection*{a}
For any policy $\pi$ and any $(s,a)$, by the definition of the action-value function and $|R_{t+1}|\leq r_{\max}$,
\[
\begin{aligned}
|q^\pi(s,a)|
&=\left|\mathbb{E}_\pi\left[\sum_{t=0}^{\infty}\gamma^t R_{t+1}\mid S_0=s,A_0=a\right]\right|\\
&\leq\mathbb{E}_\pi\left[\sum_{t=0}^{\infty}\gamma^t |R_{t+1}|\mid S_0=s,A_0=a\right]\\
&\leq r_{\max}\sum_{t=0}^{\infty}\gamma^t
=\frac{r_{\max}}{1-\gamma}.
\end{aligned}
\]
Thus,
\[
\|q^\pi\|_\infty=\max_{s,a}|q^\pi(s,a)|\leq\frac{r_{\max}}{1-\gamma}.
\]

\subsection*{b}
\noindent (i) Since $v^\star(s)=q^\star(s,\pi^\star(s))$, we have
\[
\begin{aligned}
v^\star(s)-v^{\pi_q}(s)
&=q^\star(s,\pi^\star(s))-v^{\pi_q}(s)\\
&=q^\star(s,\pi^\star(s))-q^\star(s,\pi_q(s))
 +q^\star(s,\pi_q(s))-v^{\pi_q}(s).
\end{aligned}
\]
By the Bellman equations,
\[
\begin{aligned}
q^\star(s,\pi_q(s))
&=r(s,\pi_q(s))+\gamma\sum_{s'}p(s'|s,\pi_q(s))v^\star(s'),\\
v^{\pi_q}(s)
&=r(s,\pi_q(s))+\gamma\sum_{s'}p(s'|s,\pi_q(s))v^{\pi_q}(s').
\end{aligned}
\]
Subtracting these two equations gives
\[
q^\star(s,\pi_q(s))-v^{\pi_q}(s)
=\gamma\sum_{s'}p(s'|s,\pi_q(s))
\bigl[v^\star(s')-v^{\pi_q}(s')\bigr].
\]
Therefore,
\[
\begin{aligned}
v^\star(s)-v^{\pi_q}(s)
&=q^\star(s,\pi^\star(s))-q^\star(s,\pi_q(s))\\
&\quad+\gamma\sum_{s'}p(s'|s,\pi_q(s))
\bigl[v^\star(s')-v^{\pi_q}(s')\bigr].
\end{aligned}
\]

\noindent (ii) Since $\pi_q$ is greedy with respect to $q$,
$q(s,\pi^\star(s))\leq q(s,\pi_q(s))$. Thus,
\[
\begin{aligned}
q^\star(s,\pi^\star(s))-q^\star(s,\pi_q(s))
&\leq (q^\star-q)(s,\pi^\star(s))\\
&\quad +(q-q^\star)(s,\pi_q(s))\\
&\leq 2\|q-q^\star\|_\infty.
\end{aligned}
\]
Moreover, since the transition probabilities sum to one,
\[
\sum_{s'}p(s'|s,\pi_q(s))
\bigl[v^\star(s')-v^{\pi_q}(s')\bigr]
\leq \|v^{\pi_q}-v^\star\|_\infty.
\]
Combining these inequalities with part (i), for every $s\in S$,
\[
v^\star(s)-v^{\pi_q}(s)
\leq 2\|q-q^\star\|_\infty+\gamma\|v^{\pi_q}-v^\star\|_\infty.
\]

\noindent (iii) By optimality of $v^\star$, $v^\star(s)-v^{\pi_q}(s)\geq 0$ for every $s$.
Taking the maximum over $s$ in part (ii),
\[
\|v^\star-v^{\pi_q}\|_\infty
\leq 2\|q-q^\star\|_\infty+\gamma\|v^\star-v^{\pi_q}\|_\infty.
\]
Rearranging gives
\[
\|v^\star-v^{\pi_q}\|_\infty
\leq \frac{2\|q-q^\star\|_\infty}{1-\gamma}.
\]

\subsection*{c}
Since $T_{\max}^q$ is a $\gamma$-contraction and $T_{\max}^q q^\star=q^\star$,
\[
\begin{aligned}
\|q^{(k)}-q^\star\|_\infty
&\leq \gamma^k\|q^{(0)}-q^\star\|_\infty\\
&=\gamma^k\|q^\star\|_\infty
\leq \frac{\gamma^k r_{\max}}{1-\gamma},
\end{aligned}
\]
where the last inequality follows from part (a), since $q^\star=q^{\pi^\star}$. By part (b)(iii),
\[
\|v^{\pi_{q^{(k)}}}-v^\star\|_\infty
\leq \frac{2}{1-\gamma}\|q^{(k)}-q^\star\|_\infty
\leq \frac{2r_{\max}\gamma^k}{(1-\gamma)^2}.
\]
If
\[
k\geq \left(\log\frac{1}{\gamma}\right)^{-1}
\log\frac{2r_{\max}}{(1-\gamma)^2\epsilon},
\]
then $\gamma^k\leq \frac{(1-\gamma)^2\epsilon}{2r_{\max}}$, and hence
\[
\|v^{\pi_{q^{(k)}}}-v^\star\|_\infty\leq\epsilon.
\]

\subsection*{a}
Using $|\max_a f(a)-\max_a g(a)|\leq\max_a|f(a)-g(a)|$ and
$\hat p(s'|s,a)=p(s'|s,a)$, for any value function $v$ and any $s$,
\[
\begin{aligned}
|(T_{\max}v)(s)-(\hat T_{\max}v)(s)|
&\leq \max_a |r(s,a)-\hat r(s,a)|\\
&\leq \epsilon.
\end{aligned}
\]
Hence, $\|T_{\max}v-\hat T_{\max}v\|_\infty\leq\epsilon$.
Using $v^\star=T_{\max}v^\star$ and
$\hat v^\star=\hat T_{\max}\hat v^\star$, we have
\[
\begin{aligned}
\|v^\star-\hat v^\star\|_\infty
&=\|T_{\max}v^\star-\hat T_{\max}\hat v^\star\|_\infty\\
&\leq \|T_{\max}v^\star-\hat T_{\max}v^\star\|_\infty
+\|\hat T_{\max}v^\star-\hat T_{\max}\hat v^\star\|_\infty\\
&\leq \epsilon+\gamma\|v^\star-\hat v^\star\|_\infty,
\end{aligned}
\]
where the last step uses the $\gamma$-contraction of $\hat T_{\max}$.
Thus,
\[
\|v^\star-\hat v^\star\|_\infty\leq\frac{\epsilon}{1-\gamma}.
\]

\subsection*{b}
Let $c=\epsilon/(1-\gamma)$. Since
$\hat r(s,a)=r(s,a)+\epsilon$, for every $s$,
\[
\begin{aligned}
(\hat T_{\max}(v^\star+c))(s)
&=\max_a\left\{r(s,a)+\epsilon
+\gamma\sum_{s'}p(s'|s,a)(v^\star(s')+c)\right\}\\
&=(T_{\max}v^\star)(s)+\epsilon+\gamma c\\
&=v^\star(s)+c,
\end{aligned}
\]
where we used $\sum_{s'}p(s'|s,a)=1$ and
$\epsilon+\gamma c=c$.
Therefore, $v^\star+c$ is a fixed point of $\hat T_{\max}$.
Since $\hat T_{\max}$ has a unique fixed point $\hat v^\star$,
\[
\hat v^\star(s)=v^\star(s)+\frac{\epsilon}{1-\gamma},
\qquad \forall s\in S.
\]
